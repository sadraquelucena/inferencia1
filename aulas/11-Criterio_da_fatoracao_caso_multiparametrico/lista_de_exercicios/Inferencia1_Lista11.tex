	\begin{enumerate}[label=11.\arabic*)]
		
		% Questão 1
		\item Seja $X_1,\ldots,X_n$ uma amostra aleatória da variável aleatória $X$. Encontre  estatísticas conjuntamente suficientes considerando as distribuições a seguir:
		\begin{enumerate}
			\item $X\sim$ Gama Inversa($\alpha,\beta$): $f(x) = \frac{\beta^\alpha}{\Gamma(\alpha)} x^{-(\alpha+1)} e^{-\beta/x}$, $x>0$, $\alpha,\beta>0$
			\item $X\sim$ Beta($\alpha,\beta$): $f(x) = \frac{\Gamma(\alpha+\beta)}{\Gamma(\alpha)\Gamma(\beta)} x^{\alpha-1}(1-x)^{\beta-1}$, $0<x<1$, $\alpha,\beta>0$
		\end{enumerate}
		
	\end{enumerate}
	
	\newpage
	
	\noindent Respostas:
	
	\begin{enumerate}[label=11.\arabic*)]
		
		% Questão 1
		\item 
		\begin{enumerate}
			\item $\left(\prod\limits_{i=1}^n x_i, \sum\limits_{i=1}^n x_i \right)$ são conjuntamente suficientes para $(\alpha,\beta)$.
			
			\item $\left(\prod\limits_{i=1}^n x_i, \prod\limits_{i=1}^n (1-x_i) \right)$ são conjuntamente suficientes para $(\alpha,\beta)$.
		\end{enumerate}
		
	\end{enumerate}
