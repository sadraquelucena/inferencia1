\begin{enumerate}
	\item Seja $X_1,X_2,X_3$ uma amostra aleatória independente e identicamente distribuída de uma variável aleatória $X$, com $E(X)=\mu$ e $\text{Var}(X)=\sigma^2$. Considere os seguintes estimadores para $\mu$:
	$$
	\widehat{\theta}_1=\frac{X_1+3X_2}{4} \qquad\text{e}\qquad \widehat{\theta}_2=\frac{X_1+X_2+X_3}{3}.
	$$
	\begin{enumerate}
		\item Verifique se $\widehat{\theta}_1$ é um estimador não viesado para $\mu$. \textit{(1,0 ponto)}
		
		\item Verifique se $\widehat{\theta}_2$ é um estimador não viesado para $\mu$. \textit{(1,0 ponto)}
		
		\item Calcule o erro quadrático médio (EQM) de $\widehat{\theta}_1$. \textit{(1,0 ponto)}
		
		\item Calcule o erro quadrático médio (EQM) de $\widehat{\theta}_2$. \textit{(1,0 ponto)}
		
		\item Compare os dois estimadores e indique qual é mais eficiente. Justifique sua resposta. \textit{(1,0 ponto)}
	\end{enumerate}
	
	\item Seja $X_1,\ldots,X_n$ uma amostra aleatória independente e identicamente distribuída de uma variável aleatória com distribuição Binomial Negativa, cuja função de pro\-ba\-bi\-li\-da\-de é dada por:
	$$
	f(x;\theta)=(x+1)\theta^2(1-\theta)^x, \quad 0<\theta<1, \quad x=0,1,2,\ldots
	$$
	\begin{enumerate}
		\item Apresente a função de verossimilhança $L(\theta;\mathbf{x})$. \textit{(1,5 ponto)}
		\item Utilize o critério da fatoração de Neyman para identificar uma estatística suficiente para $\theta$. \textit{(1,5 ponto)}
	\end{enumerate}
	
	\item Verifique se as distribuições a seguir pertencem à família exponencial unidimensional. Caso pertençam, apresente os elementos $c(\theta)$, $T(x)$, $d(\theta)$ e $S(x)$.
	\begin{enumerate}
		\item Distribuição Rayleigh: $f(x;\theta)=\frac{x}{\theta^2}\exp\left(-\frac{x^2}{2\theta^2}\right)$ $x>0$, $\theta>0$. \textit{(1,0 ponto)}
		\item Distribuição Cauchy: $f(x;\mu)=\frac{1}{\pi[1+(x-\mu)^2]}$, $x\in\mathbb R$, $\mu\in\mathbb R$. \textit{(1,0 ponto)}
	\end{enumerate}
\end{enumerate}
