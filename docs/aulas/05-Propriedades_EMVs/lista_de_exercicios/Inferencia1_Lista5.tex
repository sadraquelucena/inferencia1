\begin{center} {\bf Lista de Exercícios 5} \end{center}
\vspace{0.8cm}

	\begin{enumerate}[label=5.\arabic*)]
		
		% Questão 1
		\item Seja $X_1,\ldots,X_n$ uma amostra aleatória de uma distribuição $\operatorname{Exponencial}(\lambda)$, com função densidade
		\[
		f(x;\lambda)=\lambda e^{-\lambda x},\qquad x>0,\quad \lambda>0.
		\]
		Considere $\widehat{\lambda}=1/\overline{X}$, o estimador de máxima verossimilhança de $\lambda$.
		
		\begin{enumerate}[label=(\alph*)]
			\item Determine a informação de Fisher esperada para uma observação, $I_1(\lambda)$.
			
			\item Use a distribuição assintótica do EMV para obter uma aproximação para a distribuição de $\widehat{\lambda}$ quando $n$ é grande.
			
			\item Para $n=100$ e $\lambda=2$, determine aproximadamente a média e o desvio-padrão da distribuição assintótica de $\widehat{\lambda}$.
		\end{enumerate}
		
		% Questão 2
		\item Seja $X_1,\ldots,X_n$ uma amostra aleatória de uma distribuição $\operatorname{Normal}(\mu,\sigma^2)$, em que $\sigma^2$ é conhecida e $\mu$ é desconhecido. Sabe-se que o EMV de $\mu$ é $\widehat{\mu}=\overline{X}$.
		
		\begin{enumerate}[label=(\alph*)]
			\item Calcule a informação de Fisher esperada para uma observação em relação a $\mu$.
			
			\item Obtenha a distribuição assintótica de $\widehat{\mu}$ a partir da informação de Fisher.
		\end{enumerate}
		
		% Questão 3
		\item Seja $X_1,\ldots,X_n$ uma amostra aleatória de uma distribuição $\operatorname{Poisson}(\lambda)$, com $\lambda>0$. O estimador de máxima verossimilhança de $\lambda$ é $\widehat{\lambda}=\overline{X}$.
		
		Considere a quantidade
		\[
		g(\lambda)=\frac{1}{1+\lambda},
		\]
		que representa, por exemplo, uma função decrescente da intensidade média de ocorrências.
		
		\begin{enumerate}[label=(\alph*)]
			\item Determine $g'(\lambda)$.
			
			\item Obtenha o estimador de máxima verossimilhança de $g(\lambda)$.
			
			\item Usando a informação de Fisher, determine a variância assintótica de $\widehat{\lambda}$.
			
			\item Aplique o método delta para obter a distribuição assintótica de $g(\widehat{\lambda})$.
		\end{enumerate}
		
		% Questão 4
		\item Seja $X_1,\ldots,X_n$ uma amostra aleatória de uma distribuição $\operatorname{Bernoulli}(p)$, com $0<p<1$, e seja $\widehat{p}=\overline{X}$ o EMV de $p$.
		
		Considere a transformação
		\[
		g(p)=\log\left(\frac{p}{1-p}\right).
		\]
		
		\begin{enumerate}[label=(\alph*)]
			\item Determine $g'(p)$.
			
			\item Determine o EMV de $g(p)$.
			
			\item Usando a informação de Fisher para uma observação, obtenha a distribuição assintótica do estimador obtido no item (b).
			
			\item Para $p=0{,}40$ e $n=200$, calcule aproximadamente o erro-padrão assintótico desse estimador.
		\end{enumerate}
		
		% Questão 5
		\item Seja $X_1,\ldots,X_n$ uma amostra aleatória de uma distribuição $\operatorname{Gama}(\alpha,\beta)$, em que $\alpha>0$ é conhecido e $\beta>0$ é desconhecido, com parametrização pela taxa:
		\[
		f(x;\beta)=\frac{\beta^\alpha}{\Gamma(\alpha)}
		x^{\alpha-1}e^{-\beta x},\qquad x>0.
		\]
		O EMV de $\beta$ é
		\[
		\widehat{\beta}=\frac{\alpha}{\overline{X}}.
		\]
		
		\begin{enumerate}[label=(\alph*)]
			\item Determine a log-verossimilhança correspondente a uma única observação.
			
			\item Calcule a segunda derivada da log-verossimilhança em relação a $\beta$.
			
			\item Determine a informação de Fisher esperada $I_1(\beta)$.
			
			\item Obtenha a distribuição assintótica de $\widehat{\beta}$.
			
			\item Para $\alpha=4$, $\beta=2$ e $n=100$, calcule aproximadamente o erro-padrão assintótico de $\widehat{\beta}$.
		\end{enumerate}
		
		
		% Questão 6
		\item Seja $X_1,\ldots,X_n$ uma amostra aleatória de uma distribuição $\operatorname{Normal}(\mu,\sigma^2)$, com $\mu$ conhecido e $\sigma^2$ desconhecida. Considere
		\[
		f(x;\sigma^2)=
		\frac{1}{\sqrt{2\pi\sigma^2}}
		\exp\left\{
		-\frac{(x-\mu)^2}{2\sigma^2}
		\right\}.
		\]
		O estimador de máxima verossimilhança de $\sigma^2$ é
		\[
		\widehat{\sigma}^2=
		\frac{1}{n}\sum_{i=1}^n(X_i-\mu)^2.
		\]
		
		\begin{enumerate}[label=(\alph*)]
			\item Determine a informação de Fisher esperada para uma observação em relação ao parâmetro $\sigma^2$.
			
			\item Obtenha a distribuição assintótica de $\widehat{\sigma}^2$.
		\end{enumerate}
		
		
		% Questão 7
		\item Seja $X_1,\ldots,X_n$ uma amostra aleatória de uma distribuição $\operatorname{Normal}(\mu,\sigma^2)$, com $\sigma^2$ conhecida e $\mu>0$ desconhecido. O EMV de $\mu$ é $\widehat{\mu}=\overline{X}$.
		
		Considere a função
		\[
		g(\mu)=\sqrt{\mu}.
		\]
		
		\begin{enumerate}[label=(\alph*)]
			\item Determine o EMV de $g(\mu)$.
			
			\item Determine $g'(\mu)$.
			
			\item Usando o método delta, obtenha a distribuição assintótica de $\sqrt{\widehat{\mu}}$.
			
			\item Para $\mu=25$, $\sigma=5$ e $n=100$, calcule aproximadamente o erro-padrão assintótico de $\sqrt{\widehat{\mu}}$.
		\end{enumerate}
		
		% Questão 8
		\item Seja $X_1,\ldots,X_n$ uma amostra aleatória de uma distribuição $\operatorname{Poisson}(\lambda)$, com $\lambda>0$. Considere o estimador
		\[
		\widehat{\lambda}=\overline{X}.
		\]
		
		Um pesquisador deseja estimar a probabilidade de que uma unidade observada apresente pelo menos duas ocorrências:
		\[
		g(\lambda)=P(X\geq 2)
		=1-P(X=0)-P(X=1)
		=1-e^{-\lambda}(1+\lambda).
		\]
		
		\begin{enumerate}[label=(\alph*)]
			\item Determine $g'(\lambda)$.
			
			\item Usando o princípio da invariância, obtenha o estimador de máxima verossimilhança de $g(\lambda)$.
			
			\item Determine a informação de Fisher para uma observação.
			
			\item Usando o método delta, obtenha a distribuição assintótica do estimador obtido no item (b).
			
			\item Para $\lambda=1{,}5$ e $n=150$, calcule aproximadamente a variância assintótica e o erro-padrão do estimador da probabilidade $P(X\geq2)$.
		\end{enumerate}
	\end{enumerate}
	
	\newpage
	
	\noindent Respostas:
	
	\begin{enumerate}[label=5.\arabic*)]
		
		% Questão 1
		\item 
		\begin{enumerate}[label=(\alph*)]
			\item $I_1(\lambda) = \frac{1}{\lambda^2}$
			
			\item $\sqrt{n}(\widehat{\lambda} - \lambda) \xrightarrow{d} \mathcal{N}\left( 0, \frac{1}{I_1(\lambda)} \right) = \mathcal{N}(0, \lambda^2)$. \quad Equivalentemente: $\widehat{\lambda} \stackrel{a}{\sim} \mathcal{N}\left( \lambda, \frac{\lambda^2}{n} \right)$
			\item Média assintótica: $\mathbb{E}[\widehat{\lambda}] \approx \lambda = 2$\\
			      Variância assintótica: $\operatorname{Var}(\widehat{\lambda}) \approx \frac{\lambda^2}{n} = 0{,}04$\\
				  Desvio-padrão assintótico: $\operatorname{DP}(\widehat{\lambda}) = 0{,}2$
		\end{enumerate}
		
		% Questão 2
		\item 
		\begin{enumerate}
			\item $I_1(\mu) = \frac{1}{\sigma^2}$
			
			\item $\widehat{\mu} \stackrel{a}{\sim} \mathcal{N}\left( \mu, \frac{\sigma^2}{n} \right)$
		\end{enumerate}
		
		% Questão 3
		\item 
		\begin{enumerate}
			\item $g'(\lambda) = -\frac{1}{(1+\lambda)^2}$
			
			\item $\widehat{g(\lambda)} = \frac{1}{1+\overline{X}}$
			
			\item $\operatorname{Var}_{ass}(\widehat{\lambda}) = \frac{1}{n I_1(\lambda)} = \frac{\lambda}{n}$
			
			\item Pelo Método Delta, $g(\widehat{\lambda}) \stackrel{a}{\sim} \mathcal{N}\left( \frac{1}{1+\lambda}, \, \frac{\lambda}{n(1+\lambda)^4} \right)$
		\end{enumerate}
		
		% Questão 4
		\item 
		\begin{enumerate}[label=(\alph*)]
			\item $g'(p) = \frac{1}{p(1-p)}$
			
			\item $\widehat{g(p)} = \log\left( \frac{\overline{X}}{1-\overline{X}} \right)$
			
			\item $\widehat{g(p)} \stackrel{a}{\sim} \mathcal{N}\left( \log\left(\frac{p}{1-p}\right), \, \frac{1}{n \, p(1-p)} \right)$
			
			\item $\operatorname{Var}_{ass}(\widehat{g(p)}) \approx 0{,}020833$\\
			O erro-padrão assintótico é a raiz quadrada da variância assintótica: $\operatorname{EP}_{ass}(\widehat{g(p)}) \approx 0{,}1443$
		\end{enumerate}
		
		% Questão 5
		\item 
		\begin{enumerate}
			\item $\ell_1(\beta) = \alpha\ln\beta - \ln\Gamma(\alpha) + (\alpha-1)\ln x - \beta x$
			
			\item $\frac{\partial \ell_1}{\partial \beta} = \frac{\alpha}{\beta} - x \implies \frac{\partial^2 \ell_1}{\partial \beta^2} = -\frac{\alpha}{\beta^2}$
			
			\item $I_1(\beta) = \frac{\alpha}{\beta^2}$
			
			\item $\widehat{\beta} \stackrel{a}{\sim} \mathcal{N}\left( \beta, \frac{\beta^2}{n\alpha} \right)$
			
			\item $\operatorname{Var}_{ass}(\widehat{\beta}) = \frac{4}{400} = 0{,}01 \implies \operatorname{EP}_{ass}(\widehat{\beta}) = 0{,}1$
		\end{enumerate}
		
		% Questão 6
		\item 
		\begin{enumerate}
			\item $I_1(\sigma^2) = \frac{1}{2\theta^2} = \frac{1}{2(\sigma^2)^2}$
			
			\item $\widehat{\sigma}^2 \stackrel{a}{\sim} \mathcal{N}\left( \sigma^2, \frac{2(\sigma^2)^2}{n} \right)$
		\end{enumerate}
		
		% Questão 7
		\item
		\begin{enumerate}[label=(\alph*)]
			\item Pelo princípio da invariância, $\widehat{g(\mu)} = \sqrt{\overline{X}}$
			
			\item$g'(\mu) = \frac{1}{2\sqrt{\mu}}$
			
			\item Pelo método delta, $\sqrt{\widehat{\mu}} \stackrel{a}{\sim} \mathcal{N}\left( \sqrt{\mu}, \, \frac{\sigma^2}{4n\mu} \right)$
			
			\item $\operatorname{Var}_{ass}(\sqrt{\widehat{\mu}}) = 0{,}0025 \implies \operatorname{EP}_{ass}(\sqrt{\widehat{\mu}}) = 0{,}05$
		\end{enumerate}
		
		% Questão 8
		\item
		\begin{enumerate}[label=(\alph*)]
			\item $g'(\lambda) = \lambda e^{-\lambda}$
			
			\item $\widehat{g(\lambda)} = 1 - e^{-\overline{X}}(1+\overline{X})$
			
			\item $I_1(\lambda) = \frac{1}{\lambda}$
			
			\item $g(\widehat{\lambda}) \stackrel{a}{\sim} \mathcal{N}\left( 1 - e^{-\lambda}(1+\lambda), \; \frac{\lambda^3 e^{-2\lambda}}{n} \right)$
			
			\item $\operatorname{Var}_{ass} \approx 0{,}0011202 \implies \operatorname{EP}_{ass} \approx 0{,}03347$
		\end{enumerate}
		
		
	\end{enumerate}
